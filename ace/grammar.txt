primary_expr
    : IDENTIFIER
    | LITERAL
    | '(' expr ')'
    ;

dot_expr
    : primary_expr
    | dot_expr '.' primary_expr
    ;

unary_expr
    : dot_expr
    | '!' dot_expr
    | '-' dot_expr
    ;

add_expr
    : unary_expr
    | add_expr '+' unary_expr
    | add_expr '-' unary_expr
    ;

rel_expr
    : add_expr
    | rel_expr '<' add_expr
    | rel_expr '>' add_expr
    | rel_expr '<=' add_expr
    | rel_expr '>=' add_expr
    ;

eq_expr
    : rel_expr
    | eq_expr '==' rel_expr
    | eq_expr '!=' rel_expr
    | eq_expr '~' rel_expr
    | eq_expr '!~' rel_expr
    ;

and_expr
    : eq_expr
    | and_expr '&&' eq_expr
    ;

or_expr
    : and_expr
    | or_expr '||' and_expr
    ;

expr
    : or_expr
    ;
